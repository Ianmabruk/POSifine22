  EMAIL_USER=your@email.com              # For alerts
  EMAIL_PASS=yourpassword                # SMTP password
  TWILIO_ACCOUNT_SID=AC...               # Optional
  TWILIO_AUTH_TOKEN=...                  # Optional
  DATABASE_URL=postgresql://...          # If using PostgreSQL


┌─────────────────────────────────────────────────────────────────────┐
│  📡 API ENDPOINTS                                                    │
└─────────────────────────────────────────────────────────────────────┘

  GET  /api/ai/forecast                  # Sales predictions (Admin)
  POST /api/ai/pro/ask                   # AI assistant (Pro)
  GET  /api/ai/staff-score               # Employee ratings (Admin)
  POST /api/ai/alerts/check              # Manual alert check (Admin)
  GET  /api/ai/status                    # Service status (All)

┌─────────────────────────────────────────────────────────────────────┐
│  🐛 TROUBLESHOOTING                                                  │
└─────────────────────────────────────────────────────────────────────┘

  Problem: AI not working
  → Check logs: tail -f backend/logs/app.log
  → Test: python test_ai_features.py
  → Verify: curl http://localhost:5000/api/ai/status

  Problem: Frontend components not showing
  → Install: npm install recharts axios
  → Import CSS: import '../components/AIFeatures.css'
  → Check console for errors

  Problem: Alerts not sending
  → Check .env has EMAIL_USER and EMAIL_PASS
  → Test SMTP: echo "test" | mail -s "test" your@email.com
